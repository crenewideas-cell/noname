// Cache reads never wait behind a cold renderer. One background renderer yields
// to the selected preview, and all consumers share immutable resource inputs.
import {dynamicPlayerURL,dynamicThumbnailKey,compatibleThumbnailKey} from './runtime-url.js';
export function createThumbnailLoader(lookup,resources) {
 const jobs=new Map(),images=new Map(),reads=new Map(),waiting=[],targets=new WeakMap(),rows=new WeakMap(),retries=new Map(),timers=new Map(),foreground=new Set();
 let running=false,active,database;
 function db(){return database||=new Promise(resolve=>{let settled=false;const finish=value=>{if(settled){value?.close();return;}settled=true;clearTimeout(timer);resolve(value);};const timer=setTimeout(()=>finish(null),700);try{const r=indexedDB.open('noname-dynamic-thumbnails',1);r.onupgradeneeded=()=>{if(!r.result.objectStoreNames.contains('images'))r.result.createObjectStore('images');};r.onsuccess=()=>finish(r.result);r.onerror=r.onblocked=()=>finish(null);}catch{finish(null);}});}
 async function cached(key,value){const d=await db();if(!d)return;return new Promise(resolve=>{const timer=setTimeout(resolve,700),finish=v=>{clearTimeout(timer);resolve(v);};try{const t=d.transaction('images',value?'readwrite':'readonly'),r=value?t.objectStore('images').put(value,key):t.objectStore('images').get(key);r.onsuccess=()=>finish(r.result);r.onerror=t.onabort=()=>finish();}catch{finish();}});}
 const visible=node=>{if(!node.isConnected||!node.getClientRects().length)return false;const r=node.getBoundingClientRect(),c=node.closest('.qh-skinchange-shousha-big-skin')?.parentElement?.parentElement.getBoundingClientRect()||{left:0,top:0,right:innerWidth,bottom:innerHeight};return r.right>Math.max(0,c.left)&&r.left<Math.min(innerWidth,c.right)&&r.bottom>Math.max(0,c.top)&&r.top<Math.min(innerHeight,c.bottom);};
 function remember(key,source){images.delete(key);images.set(key,source);if(images.size>160)images.delete(images.keys().next().value);retries.delete(key);void cached(key,source);}
 function apply(row,source){for(const node of row.nodes)if(targets.get(node)===row.key&&node.isConnected){node.style.backgroundImage='url('+JSON.stringify(source)+')';node.dataset.skinThumbnailReady='true';delete node.dataset.skinThumbnailError;}row.nodes.clear();}
 async function read(own){const key=dynamicThumbnailKey(own.p,own.e);if(images.has(key))return images.get(key);if(!reads.has(key)){const promise=(async()=>{const result=await cached(key)||await cached(compatibleThumbnailKey(own.p,own.e));if(result)remember(key,result);return result;})().finally(()=>reads.delete(key));reads.set(key,promise);}return reads.get(key);}
 async function snapshot(row){const frame=document.createElement('iframe'),channel='thumbnail-'+crypto.randomUUID();frame.__nonameSkinEntry=row.e;frame.__nonameSkinResources=resources;
  frame.style.cssText='position:fixed;left:-1000px;top:0;width:240px;height:360px;border:0;pointer-events:none';frame.setAttribute('aria-hidden','true');frame.dataset.skinThumbnail='true';
  try{return await new Promise((resolve,reject)=>{let finished=false;const finish=(error,value)=>{if(finished)return;finished=true;clearTimeout(timer);clearInterval(abandon);removeEventListener('message',message);row.finish=undefined;error?reject(error):resolve(value);};row.finish=finish;
   const timer=setTimeout(()=>finish(Error('缩略图渲染超时')),15000),abandon=setInterval(()=>{if(![...row.nodes].some(visible))finish(Object.assign(Error('缩略图已不可见'),{deferred:true}));},400);
   const message=event=>{if(event.source!==frame.contentWindow||event.data?.channel!==channel)return;if(event.data.type==='noname-skin-error')finish(Error(event.data.message));if(event.data.type==='noname-skin-ready')try{const p=frame.contentWindow.skinPlayer;p.pause(true);p.renderFrame?.();finish(null,p.canvas.toDataURL('image/webp',.82));}catch(e){finish(e);}};
   addEventListener('message',message);frame.src=dynamicPlayerURL(row.p.base,{id:row.e.id,presentation:'portrait',channel});document.body.append(frame);
  });}finally{frame.contentWindow?.skinPlayerLifecycle?.dispose();frame.remove();}}
 const observer=new IntersectionObserver(entries=>{for(const e of entries){if(!e.target.isConnected){observer.unobserve(e.target);continue;}if(e.isIntersecting){observer.unobserve(e.target);request(rows.get(e.target),e.target);}}});
 function schedule(row,error){const attempts=error?.deferred?(retries.get(row.key)||0):(retries.get(row.key)||0)+1;retries.set(row.key,attempts);const nodes=[...row.nodes].filter(n=>targets.get(n)===row.key&&n.isConnected);if(!nodes.length)return;
  for(const node of nodes){if(!error?.deferred)node.dataset.skinThumbnailError='retry';if(!visible(node))observer.observe(node);}
  if(attempts<=2){clearTimeout(timers.get(row.key));timers.set(row.key,setTimeout(()=>{timers.delete(row.key);for(const node of nodes)if(targets.get(node)===row.key&&visible(node))request(rows.get(node),node);},error?.deferred?250:attempts*700));}
 }
 async function pump(){if(running||foreground.size)return;running=true;try{while(waiting.length&&!foreground.size){const row=waiting.shift();if(jobs.get(row.key)!==row)continue;let error;try{if(![...row.nodes].some(visible)){error={deferred:true};continue;}active=row;const source=images.get(row.key)||await snapshot(row);remember(row.key,source);apply(row,source);}catch(e){error=e;}finally{active=undefined;if(jobs.get(row.key)===row)jobs.delete(row.key);if(error)schedule(row,error);}await new Promise(resolve=>setTimeout(resolve,50));}}finally{running=false;if(waiting.length&&!foreground.size)void pump();}}
 function enqueue(own,node){if(!own)return;const key=dynamicThumbnailKey(own.p,own.e);if(targets.get(node)!==key)return;if(images.has(key)){apply({key,nodes:new Set([node])},images.get(key));return;}let row=jobs.get(key);if(!row){row={...own,key,nodes:new Set()};jobs.set(key,row);waiting.push(row);}row.nodes.add(node);void pump();}
 async function request(own,node){if(!own)return;const key=dynamicThumbnailKey(own.p,own.e),source=await read(own);if(targets.get(node)!==key||!node.isConnected)return;if(source)apply({key,nodes:new Set([node])},source);else if(visible(node))enqueue(own,node);else observer.observe(node);}
 function watch(own,node){const key=dynamicThumbnailKey(own.p,own.e);if(targets.get(node)!==key){delete node.dataset.skinThumbnailReady;delete node.dataset.skinThumbnailError;}rows.set(node,own);targets.set(node,key);if(!node.style.backgroundImage||node.style.backgroundImage==='none')node.style.backgroundImage='url('+JSON.stringify(own.p.base+own.e.thumbnail)+')';if(visible(node))void request(own,node);else observer.observe(node);}
 const load=(name,skin,node)=>{const own=lookup(name,skin);if(!own||!own.e.thumbnail?.endsWith('.svg'))return false;watch(own,node);return true;};
 load.preview=(p,e,node)=>{const own={p,e},key=dynamicThumbnailKey(p,e);rows.set(node,own);targets.set(node,key);if(e.thumbnail&&!e.thumbnail.endsWith('.svg')){apply({key,nodes:new Set([node])},p.base+e.thumbnail);return;}if(images.has(key))apply({key,nodes:new Set([node])},images.get(key));else void read(own).then(source=>{if(source)apply({key,nodes:new Set([node])},source);});};
 load.prioritize=()=>{const token={};foreground.add(token);active?.finish?.(Object.assign(Error('优先加载选中预览'),{deferred:true}));let released=false;return()=>{if(released)return;released=true;foreground.delete(token);void pump();};};
 load.publish=(p,e,player)=>{if(!player?.canvas)return;const key=dynamicThumbnailKey(p,e);if(images.has(key))return;try{player.renderFrame?.();const c=document.createElement('canvas');c.width=240;c.height=360;const scale=Math.max(c.width/player.canvas.width,c.height/player.canvas.height),w=player.canvas.width*scale,h=player.canvas.height*scale;c.getContext('2d').drawImage(player.canvas,(c.width-w)/2,(c.height-h)/2,w,h);const source=c.toDataURL('image/webp',.82);remember(key,source);const job=jobs.get(key);if(job){apply(job,source);job.finish?.(null,source);}}catch{}};
 load.cancel=node=>{targets.delete(node);rows.delete(node);observer.unobserve(node);for(const row of jobs.values())row.nodes.delete(node);};
 load.stats=()=>({queued:waiting.filter(r=>jobs.get(r.key)===r).length,running:!!active,foreground:foreground.size,images:images.size,cacheReads:reads.size});
 return load;
}
